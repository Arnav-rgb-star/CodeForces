import java.io.*;
import java.util.*;
public class deliveryDilemma{
    public static void main(String[] args)throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringBuilder result = new StringBuilder();
        int t = Integer.parseInt(br.readLine());

        while(t-->0){
            int n = Integer.parseInt(br.readLine());
            StringTokenizer st = new StringTokenizer(br.readLine());

            int[][] v = new int[n][2];
            for(int i=0;i<n;i++) v[i][0] = Integer.parseInt(st.nextToken());
            st = new StringTokenizer(br.readLine());
            
            long suffix[]=new long[n+1];
            
            for(int i=0;i<n;i++) v[i][1]=Integer.parseInt(st.nextToken());
            
            Arrays.sort(v,(x,y)->Integer.compare(x[0],y[0]));
            
            suffix[n-1]=v[n-1][1];
            for(int i=n-2;i>=0;i--) suffix[i]=suffix[i+1]+v[i][1];
            long max=0;
            long time=0;
            long min = suffix[0];

            for(int i=0;i<n;i++){
                max = v[i][0];
                time = Math.max(max,suffix[i+1]);
                min = Math.min(time,min);
            }
            result.append(min).append("\n");
        }
 
        System.out.println(result);
    }
}
