import java.io.*;
import java.util.*;
public class redBlueBeans{
    public static void main(String[] args)throws IOException {
       Scanner in = new Scanner(System.in);
        StringBuilder result = new StringBuilder();
        int t = in.nextInt();

        while(t-->0){
            
            int r = in.nextInt();
            int b = in.nextInt();
            int d = in.nextInt();

            if(1L*Math.max(r,b)<=Math.min(r,b)*1L*(d+1)) result.append("Yes").append("\n");
            else result.append("No").append("\n");
        }
 
        System.out.println(result);
    }
}
