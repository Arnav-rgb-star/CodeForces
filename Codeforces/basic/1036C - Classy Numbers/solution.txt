ll dp[20][2][2][11];
//basic digit dp with 4 states
ll dfs(int pos,int tight,int leadingZero,int nonZerosSoFar){
    if(nonZerosSoFar>3)return dp[pos][tight][leadingZero][nonZerosSoFar]=0;
    if(pos==((int)digits.size()))return (nonZerosSoFar<=3);

    if(dp[pos][tight][leadingZero][nonZerosSoFar]!=-1)return dp[pos][tight][leadingZero][nonZerosSoFar];
    ll ans=0;
    int limit=tight?digits[pos]:9;
    for(int d=0;d<=limit;d++){

        if(!leadingZero){//number has started
            // if(d==lastDigit)continue;
            if(nonZerosSoFar>3)return dp[pos][tight][leadingZero][nonZerosSoFar]=0;
        }

        int nextTight=tight&&(d==limit);
        int nextLeadingZero=leadingZero&&(d==0);
        int nextnonZeros=nonZerosSoFar+(d!=0);
        // if(nextLeadingZero)nextLastDigit=10;//which isnt in range
        // else nextLastDigit=d;
        ans+=dfs(pos+1,nextTight,nextLeadingZero,nextnonZeros);
    }
    return dp[pos][tight][leadingZero][nonZerosSoFar]=ans;
}

ll count(ll x){
    if(x<0)return 0;
    digits.clear();
    while(x>0){digits.pb(x%10);x/=10;}
    reverse(all(digits));
    memset(dp,-1,sizeof(dp));
    return dfs(0,1,1,0);//tight=1,leadingZeros=1,lastDigit=10
}
