#include <iostream>
using namespace std;

int main() {
    int t;
    cin >> t;

    while (t--) {
        int n;
        cin >> n;

        int a[1005];
        int totalTwos = 0;

        for (int i = 0; i < n; i++) {
            cin >> a[i];
            if (a[i] == 2) {
                totalTwos++;
            }
        }

        if (totalTwos % 2 == 1) {
            cout << -1 << endl;
            continue;
        }

        if (totalTwos == 0) {
            cout << 1 << endl;
            continue;
        }

        int need = totalTwos / 2;
        int countTwos = 0;

        for (int i = 0; i < n; i++) {
            if (a[i] == 2) {
                countTwos++;
            }

            if (countTwos == need) {
                cout << i + 1 << endl;
                break;
            }
        }
    }

    return 0;
}
