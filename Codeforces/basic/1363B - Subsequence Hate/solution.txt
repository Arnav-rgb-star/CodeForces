import java.io.*;
import java.util.*;
public class subHate{
    public static void main(String[] args)throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringBuilder result = new StringBuilder();
        int t = Integer.parseInt(br.readLine());

        while(t-->0){
            String s = br.readLine();

            int n = s.length();
            int prefix0[] = new int[n];
            int prefix1[] = new int[n];

            int suffix0[] = new int[n];
            int suffix1[] = new int[n];


            if(s.charAt(0)=='1') prefix1[0]=1;
            else prefix0[0]=1;

            for(int i=1;i<n;i++){
                prefix0[i]= prefix0[i-1];
                prefix1[i]= prefix1[i-1];

                if(s.charAt(i)== '0') prefix0[i]++;
                else prefix1[i]++;
            }

            if(s.charAt(n-1)=='1') suffix1[n-1]=1;
            else suffix0[n-1]=1;

            for(int i=n-2;i>=0;i--){

                suffix1[i]= suffix1[i+1];
                suffix0[i]= suffix0[i+1];

                if(s.charAt(i)=='0') suffix0[i]++;
                else suffix1[i]++;
            }

            int a=prefix1[n-1];
            for(int i=0;i<n-1;i++){
                a = Math.min(a,prefix1[i]+suffix0[i+1]);
            }

            int b=prefix0[n-1];
            for(int i=0;i<n-1;i++){
                b = Math.min(b,prefix0[i]+suffix1[i+1]);
            }
            
            result.append(Math.min(a,b)).append("\n");
        }
 
        System.out.println(result);
    }
}
