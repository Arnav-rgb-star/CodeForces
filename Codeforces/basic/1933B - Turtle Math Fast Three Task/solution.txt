t = int(input())
for _ in range(t):
    n = int(input())
    a = list(map(int, input().split()))
    s = sum(a)
    r = s % 3
    if r == 0:
        print(0)
    elif r == 2:
        print(1)
    else:
        ok = False
        for i in a:
            if i % 3 == 1:
                ok = True
                break
        if ok:
            print(1)
        else:
            print(2)
