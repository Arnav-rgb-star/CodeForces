import java.io.*;
import java.util.*;

public class riskyTower{
    public static void main(String[] args)throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringBuilder result = new StringBuilder();
        int t = Integer.parseInt(br.readLine());

        while(t-->0){
            StringTokenizer st = new StringTokenizer(br.readLine());
            int n = Integer.parseInt(st.nextToken());
            int m = Integer.parseInt(st.nextToken());

            long a[][]=new long[n][m];
            long min[]=new long[n];
            long sum[]=new long[n];
        
            long ops=m;

            st = new StringTokenizer(br.readLine());
            long v[]=new long[n];
            for(int i=0;i<n;i++) v[i]=Integer.parseInt(st.nextToken());

            for(int i=0;i<n;i++){
                st = new StringTokenizer(br.readLine());
                for(int j=0;j<m;j++){
                    a[i][j]=Integer.parseInt(st.nextToken());
                }
            }

            min[0]=v[0];
            for(int i=1;i<n;i++) min[i]=Math.min(min[i-1],v[i]);

            PriorityQueue<Long> pq = new PriorityQueue<>();

            for(int i=n-1;i>=0;i--){

                for(int j=0;j<m;j++){
                    pq.offer(a[i][j]);
                    if(pq.size()>m) pq.poll();           
                }
                int k=0;
                sum[i]=0;

                for(long value : pq){
                    a[i][k++]=value;
                    sum[i]+=value;
                }
                Arrays.sort(a[i]);
            }

            for(int i=n-1;i>=0;i--){
                if(sum[i]>=min[i]){
                    long temp=0;

                    for(int j=m-1;j>=0;j--){
                        temp+=a[i][j];

                        if(temp>=min[i]){
                            ops=Math.min(ops,m-j);
                            break;
                        }
                    }
                }
            }
            result.append(ops).append("\n");
        }
 
        System.out.println(result);
    }
}
