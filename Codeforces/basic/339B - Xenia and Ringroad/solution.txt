#include <iostream>
#include <vector>
#include <map>
#include<set>
#include <unordered_set>
#include<cmath>
#include <algorithm>
using namespace std;
int main(){
    int n,m;
    cin>>n>>m;

    long long ans=0;
    int curr=1;

    while(m--){
        int a;
        cin>>a;

        if (a>=curr)
            ans+=a-curr;
        else
            ans+=(n-curr)+a;
        curr=a;
    }
    cout<<ans<<endl;
    return 0;
}
